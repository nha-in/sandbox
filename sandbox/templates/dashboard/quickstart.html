Python 
import os, httpx

response = httpx.post(
    "{{ gateway_base_url }}/v0.5/sessions",
    json={
        "clientId": "{{ credentials.client_id }}",
        "clientSecret": os.environ["ABDM_CLIENT_SECRET"],
    },
)
response.raise_for_status()

const response = await fetch("{{ gateway_base_url }}/v0.5/sessions", {
  method: "POST",

  body: JSON.stringify({
    clientId: "{{ credentials.client_id }}",
    clientSecret: process.env.ABDM_CLIENT_SECRET,
  }),
});
